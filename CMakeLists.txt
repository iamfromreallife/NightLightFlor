cmake_minimum_required(VERSION 3.23)
project(NightLightNative VERSION 1.0.15 LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "Native NightLight requires Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    # Windows MSVC otherwise reads UTF-8 source as the machine's ACP and
    # displays Ã§/Ã£/Ãª instead of ç/ã/ê in wide Portuguese strings.
    add_compile_options(/utf-8)
endif()

add_library(nightlight-core STATIC core.cpp display_identity.cpp ddc.cpp wmi_brightness.cpp)
target_include_directories(nightlight-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(nightlight-core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(nightlight-core PUBLIC user32 gdi32 shell32 ole32 oleaut32 wbemuuid advapi32 dxva2)

set(ICON_PATH "${CMAKE_CURRENT_SOURCE_DIR}/icon.ico")
set(ICON_DARK_PATH "${CMAKE_CURRENT_SOURCE_DIR}/icon-dark.ico")
set(ICON_LIGHT_PATH "${CMAKE_CURRENT_SOURCE_DIR}/icon-light.ico")
if(NOT EXISTS "${ICON_PATH}" OR NOT EXISTS "${ICON_DARK_PATH}" OR NOT EXISTS "${ICON_LIGHT_PATH}")
    message(FATAL_ERROR "Missing application or tray icon resource.")
endif()
configure_file(resources.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)

add_executable(NightLight WIN32 main.cpp startup_task.cpp inactive_media.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
target_link_libraries(NightLight PRIVATE nightlight-core
    user32 gdi32 shell32 ole32 advapi32 comctl32 dwmapi uxtheme gdiplus taskschd uuid)
target_compile_definitions(NightLight PRIVATE UNICODE _UNICODE)
if(MSVC)
    target_compile_options(NightLight PRIVATE /W4 /permissive- /EHsc)
    # Modern C++/WinRT uses standard coroutines, not deprecated /await.
    target_compile_features(NightLight PRIVATE cxx_std_20)
    target_link_libraries(NightLight PRIVATE windowsapp)
    target_link_options(NightLight PRIVATE /MANIFEST:EMBED)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(nightlight-native-tests tests.cpp)
    target_link_libraries(nightlight-native-tests PRIVATE nightlight-core)
    if(MSVC)
        target_compile_options(nightlight-native-tests PRIVATE /W4 /permissive- /EHsc)
    endif()
    add_test(NAME native-regressions COMMAND nightlight-native-tests)
endif()
